# IP-Spoof-test - Manual system requirements
# Repository: KindReYX/IP-Spoof-test
#
# This is NOT a pip-only requirements file.
# The project depends primarily on Linux system networking tools.
# Use ONE section below for your distribution.
# Run installation commands as root or with sudo.

============================================================
Debian / Ubuntu
============================================================

sudo apt update
sudo apt install -y \
  bash \
  coreutils \
  curl \
  iproute2 \
  iptables \
  nftables \
  iputils-ping \
  tcpdump \
  conntrack \
  nmap \
  openssh-client \
  sshpass \
  python3 \
  python3-pip \
  python3-scapy \
  screen

============================================================
Fedora / Rocky Linux / AlmaLinux / RHEL-compatible
============================================================

sudo dnf install -y \
  bash \
  coreutils \
  curl \
  iproute \
  iptables \
  nftables \
  iputils \
  tcpdump \
  conntrack-tools \
  nmap \
  openssh-clients \
  sshpass \
  python3 \
  python3-pip \
  screen

# If dnf is unavailable but yum exists:
# sudo yum install -y bash coreutils curl iproute iptables nftables iputils tcpdump conntrack-tools nmap openssh-clients sshpass python3 python3-pip screen

# Scapy package availability varies by distribution/version.
# Prefer the distro package if available. Otherwise use a Python virtual environment.
